Reactive UI on the DOM.

Build a page or embed a widget with typed elements, signals and Web Components.

Get startedGitHub
npm create ecmact@latest my-app
main.ts
import { mount, signal, tags } from 'ecmact';

const hours = signal(1);
const estimate = () => `$${hours.value * 49}`;

mount('#app',
  tags.button(
    { onclick: () => hours.value++ },
    'Add an hour',
  ),
  tags.output(estimate),
);
